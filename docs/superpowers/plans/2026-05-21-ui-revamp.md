> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the current generic dark SaaS aesthetic with the spec-defined "noir editorial" design system — new color palette, custom fonts (Schibsted Grotesk / Newsreader / IBM Plex Mono), redesigned components, and updated screens — while keeping all backend wiring intact.

### Task 5: Rebuild Text component with new font system

**Files:**
- Modify: `components/ui/Text.tsx`

- [ ] **Step 1: Replace `components/ui/Text.tsx`**

The new Text component supports `family` prop to select font family directly, handles the italic-serif discipline (only `serifItalic` for wordmark/display numerals), and defaults to Schibsted Grotesk.

```tsx
import React from 'react';
import { Text as RNText, TextProps as RNTextProps } from 'react-native';
import { colors } from '@/theme/colors';
import { typography, getScaledFontSizes } from '@/theme/typography';

type FontFamily =
  | 'sans'          // Schibsted Grotesk 400 — default UI
  | 'sansSemibold'  // Schibsted Grotesk 600 — headings, CTAs
  | 'sansBold'      // Schibsted Grotesk 700
  | 'serif'         // Newsreader upright — reading-mode entries
  | 'serifItalic'   // Newsreader italic — wordmark, display numerals ≥ 40px only
  | 'serifSemiboldItalic' // Newsreader 600 italic — milestone numbers
  | 'mono'          // IBM Plex Mono 400 — metadata, eyebrows
  | 'monoMedium';   // IBM Plex Mono 500 — counters

interface TextProps extends RNTextProps {
  variant?: keyof ReturnType<typeof getScaledFontSizes>;
  family?: FontFamily;
  /** Shorthand: serif upright (reading mode) */
  serif?: boolean;
  /** Shorthand: mono metadata */
  mono?: boolean;
  color?: 'primary' | 'bright' | 'secondary' | 'tertiary' | 'accent' | 'warn';
  centered?: boolean;
  fixedSize?: boolean;
  uppercase?: boolean;
}

export function Text({
  style,
  variant = 'base',
  family,
  serif,
  mono,
  color = 'primary',
  centered,
  fixedSize = false,
  uppercase,
  allowFontScaling = true,
  ...props
}: TextProps) {
  const activeColors = colors.dark;
  const scaledSizes = getScaledFontSizes();
  const fontSize = fixedSize ? typography.fontSize[variant as keyof typeof typography.fontSize] ?? 16 : (scaledSizes[variant] ?? 16);

  const getFontFamily = (): string => {
    if (family) return typography.fontFamily[family];
    if (mono) return typography.fontFamily.mono;
    if (serif) return typography.fontFamily.serif;
    return typography.fontFamily.sans;
  };

  const getTextColor = (): string => {
    const map: Record<string, string> = {
      primary: activeColors.textPrimary,
      bright: activeColors.textBright,
      secondary: activeColors.textSecondary,
      tertiary: activeColors.textTertiary,
      accent: activeColors.accent,
      warn: activeColors.warn,
    };
    return map[color] ?? activeColors.textPrimary;
  };

  return (
    <RNText
      style={[
        {
          fontSize,
          fontFamily: getFontFamily(),
          color: getTextColor(),
          textAlign: centered ? 'center' : 'auto',
          textTransform: uppercase ? 'uppercase' : undefined,
        },
        style,
      ]}
      allowFontScaling={allowFontScaling && !fixedSize}
      {...props}
    />
  );
}
```

import React from 'react';
import { TouchableOpacity, ActivityIndicator, StyleSheet, ViewStyle } from 'react-native';
import { Text } from './Text';
import { colors } from '@/theme/colors';
import { componentRadius, spacing } from '@/theme/spacing';
import * as Haptics from 'expo-haptics';

interface ButtonProps {
  title: string;
  onPress: () => void;
  variant?: 'primary' | 'ghost' | 'outline';
  disabled?: boolean;
  loading?: boolean;
  style?: ViewStyle;
}

export function Button({ title, onPress, variant = 'primary', disabled, loading, style }: ButtonProps) {
  const c = colors.dark;

  const handlePress = () => {
    if (disabled || loading) return;
    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
    onPress();
  };

  const bg = variant === 'primary'
    ? (disabled ? c.frame : c.textBright)
    : 'transparent';

  const textColor = variant === 'primary'
    ? (disabled ? c.textTertiary : c.paper)  // dark text on white pill
    : (disabled ? c.textTertiary : c.textPrimary);

  const borderColor = variant === 'outline' ? (disabled ? c.textTertiary : c.textPrimary) : 'transparent';

  return (
    <TouchableOpacity
      onPress={handlePress}
      disabled={disabled || loading}
      activeOpacity={0.75}
      style={[
        styles.base,
        {
          backgroundColor: bg,
          borderColor,
          borderWidth: variant === 'outline' ? 1 : 0,
          opacity: disabled ? 0.5 : 1,
        },
        style,
      ]}
    >
      {loading
        ? <ActivityIndicator color={textColor} size="small" />
        : <Text family="sansSemibold" variant="base" style={{ color: textColor }}>{title}</Text>
      }
    </TouchableOpacity>
  );
}

const styles = StyleSheet.create({
  base: {
    height: 52,
    borderRadius: componentRadius.button,
    paddingHorizontal: spacing[6],
    alignItems: 'center',

import React from 'react';
import { View } from 'react-native';
import Svg, { Rect, Circle } from 'react-native-svg';
import { colors } from '@/theme/colors';

interface LogoProps {
  size?: number;
  /** 'primary' = white tile ink dots (for dark surfaces, default)
   *  'reversed' = ink tile bone dots (for light surfaces) */
  variant?: 'primary' | 'reversed';
}

export function Logo({ size = 64, variant = 'primary' }: LogoProps) {
  const tileRadius = size * 0.22;
  const dotR = size * 0.062;
  const dotSpacing = size * 0.21;
  const cx = size / 2;
  const cy = size / 2;

  const tileFill = variant === 'primary' ? colors.dark.textBright : colors.dark.paper;
  const dotFill = variant === 'primary' ? colors.dark.paper : colors.dark.textPrimary;

  return (
    <View accessibilityLabel="Three Sentences logo">
      <Svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
        <Rect
          x={0}
          y={0}
          width={size}
          height={size}
          rx={tileRadius}
          ry={tileRadius}
          fill={tileFill}
        />
        <Circle cx={cx - dotSpacing} cy={cy} r={dotR} fill={dotFill} />
        <Circle cx={cx}             cy={cy} r={dotR} fill={dotFill} />
        <Circle cx={cx + dotSpacing} cy={cy} r={dotR} fill={dotFill} />
      </Svg>
    </View>
  );
}

import React, { useEffect, useRef } from 'react';
import { StyleSheet, Animated, Pressable, View } from 'react-native';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import { Text } from '@/components/ui/Text';
import { colors } from '@/theme/colors';
import { spacing } from '@/theme/spacing';

export type ToastType = 'success' | 'error' | 'info';

interface ToastProps {
  type: ToastType;
  message: string;
  visible: boolean;
  duration?: number;
  onDismiss: () => void;
}

export function Toast({ type, message, visible, duration = 3000, onDismiss }: ToastProps) {
  const insets = useSafeAreaInsets();
  const translateY = useRef(new Animated.Value(-80)).current;
  const opacity = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    if (visible) {
      Animated.parallel([
        Animated.timing(translateY, {
          toValue: 0,
          duration: 280,
          useNativeDriver: true,
        }),
        Animated.timing(opacity, {
          toValue: 1,
          duration: 200,
          useNativeDriver: true,
        }),
      ]).start();

      const timer = setTimeout(() => {
        hide();
      }, duration);
      return () => clearTimeout(timer);
    }
  }, [visible]);

  const hide = () => {
    Animated.parallel([
      Animated.timing(translateY, {
        toValue: -80,
        duration: 200,
        useNativeDriver: true,
      }),
      Animated.timing(opacity, {
        toValue: 0,
        duration: 180,
        useNativeDriver: true,
      }),
    ]).start(() => onDismiss());
  };

  const borderColor = type === 'error' ? colors.dark.warn : colors.dark.frame;

  return (
    <Animated.View
      style={[
        styles.container,
        {
          top: insets.top + spacing[3],
          transform: [{ translateY }],
          opacity,
          borderColor,
        },
      ]}
      pointerEvents="box-none"

  status,
  onPress,
  disabled,

  status: 'idle' | 'submitting' | 'success';
  onPress: () => void;
  disabled: boolean;

  const c = colors.dark;

        Animated.timing(widthAnim, { toValue: 0, duration: 300, useNativeDriver: false }),
        Animated.timing(spinValue, { toValue: 1, duration: 400, useNativeDriver: false }),
        Animated.timing(fadeValue, { toValue: 0, duration: 400, delay: 800, useNativeDriver: false }),
      ]).start();
    } else if (status === 'idle') {

      spinValue.setValue(0);
      fadeValue.setValue(1);
    }
  }, [status]);

  const spin = spinValue.interpolate({ inputRange: [0, 1], outputRange: ['0deg', '360deg'] });
  const animatedWidth = initialWidth
    ? widthAnim.interpolate({ inputRange: [0, 1], outputRange: [56, initialWidth] })
    : '100%';

  const isEnabled = !disabled && status === 'idle';

  return (

      onLayout={(e) => { if (!initialWidth) setInitialWidth(e.nativeEvent.layout.width); }}
